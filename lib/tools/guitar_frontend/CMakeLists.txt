cmake_minimum_required(VERSION 3.22)
project(GuitarFrontend VERSION 0.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CLAP_INCLUDE_DIR "" CACHE PATH "Pinned CLAP include directory (see ../chordlab_clap/dependencies.toml)")
if(NOT EXISTS "${CLAP_INCLUDE_DIR}/clap/clap.h")
    message(FATAL_ERROR "Set CLAP_INCLUDE_DIR to the already verified CLAP headers")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS defaults.toml configure.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/configure.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/defaults.toml" "${CMAKE_CURRENT_BINARY_DIR}/defaults.h" COMMAND_ERROR_IS_FATAL ANY)
add_library(guitar_engine STATIC engine.cpp)
target_include_directories(guitar_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
add_library(GuitarFrontend MODULE plugin.cpp)
target_include_directories(GuitarFrontend PRIVATE "${CLAP_INCLUDE_DIR}")
target_link_libraries(GuitarFrontend PRIVATE guitar_engine)
set_target_properties(GuitarFrontend PROPERTIES PREFIX "" SUFFIX ".clap" OUTPUT_NAME "Guitar Frontend")
foreach(t guitar_engine GuitarFrontend)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
enable_testing()
add_executable(engine_tests engine_tests.cpp)
target_link_libraries(engine_tests PRIVATE guitar_engine)
add_test(NAME detector COMMAND engine_tests)
add_executable(host_tests host_tests.cpp)
target_include_directories(host_tests PRIVATE "${CLAP_INCLUDE_DIR}")
target_link_libraries(host_tests PRIVATE ${CMAKE_DL_LIBS})
add_test(NAME clap_host COMMAND host_tests "$<TARGET_FILE:GuitarFrontend>")

# Offline diagnostic only; raw native float64 at 48 kHz on stdin.
add_executable(replay_clip replay_clip.cpp)
target_link_libraries(replay_clip PRIVATE guitar_engine)

# Keep assertion-based tests active even in Release builds.
target_compile_options(engine_tests PRIVATE -UNDEBUG)
